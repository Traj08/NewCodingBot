1) Install git in your m/c
2) Create an account in github(web)
3) Create a repository in the web(remote repository)

4) create a folder in the system

-- Adding a file in the Local repo:
Admin@Disney MINGW64 /d/2025-Apr-Wipro/mygit/aprwipro (main)
$ echo Good Evening > file1.txt    -- file1.txt got created in the working area.

git status
git add file1.txt		-- Push the changes in the Working directory to the statging area

git status
git commit -m "First Commit"    -- The changes are made permanent in the local repository / git repo

-- Edit the file
git status
git add file1.txt
git status
 git commit -m "Changed Evening to Morning"

-- If multiple files are created / edited
echo Hi > f1.txt
 echo Hello > f2.txt
 git status
 git add .
 git status
git commit -m "Created f1.txt and f2.txt"

-- Rename the file
 mv f2.txt f3.txt
git status
git add .
 git status
 git commit -m "Renamed f2.txt f3.txt"

------------
 git mv f3.txt f4.txt
git status
 git commit -m "f3.txt to f4.txt"


-------------------
-- Remove a file
 git rm f2.txt
git status
 git commit -m "Deleted f2.txt"

xy

x - status in staging area(index)
y - unstaged changes

Admin@Disney MINGW64 /d/2025-Apr-Wipro/mygit/aprwipro (main)
$ git status -s
A_  f5.txt		-- A for Added and staged

$ git commit -m "Created f5.txt"

------------------------------
Editing the file
$ git status -s
_M f5.txt		-- Modified but not staged . It is in working directory
$ git add .
$ git status -s
M_  f5.txt		-- Modified but statged

$ git commit -m "modified f5.txt"

-----------------------------
Renamed
$ git mv f5.txt f6.txt
$ git status -s
R  f5.txt -> f6.txt
$ git commit -m "Renamed f5.txt f6.txt"


--------------------------------
Copy contents / file 
 cp f6.txt f7.txt
$ git status -s
?? f7.txt
$ git add .
$ git status -s
A_  f7.txt
$ git commit -m "f7.txt created from f6.txt"
